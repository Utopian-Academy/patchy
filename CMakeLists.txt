cmake_minimum_required(VERSION 3.24)
project(Patchy VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_library(patchy_core src/core/format.cpp src/core/bridge.cpp src/core/channel_map.cpp)
target_include_directories(patchy_core PUBLIC include)
target_compile_features(patchy_core PUBLIC cxx_std_20)
set_target_properties(patchy_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(patchy_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(patchy_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Patchy initially targets Windows x64")
  endif()
  add_library(patchy_windows src/windows/devices.cpp src/windows/router.cpp)
  target_include_directories(patchy_windows PUBLIC include PRIVATE src/windows)
  target_compile_definitions(patchy_windows PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(patchy_windows PUBLIC patchy_core PRIVATE ole32 uuid avrt propsys)
  add_executable(patchy src/cli/main.cpp)
  target_include_directories(patchy PRIVATE src/windows)
  target_link_libraries(patchy PRIVATE patchy_windows)
  if(MSVC)
    target_compile_options(patchy_windows PRIVATE /W4 /permissive- /utf-8)
    target_compile_options(patchy PRIVATE /W4 /permissive- /utf-8)
  endif()
  install(TARGETS patchy RUNTIME DESTINATION bin)
else()
  message(STATUS "Non-Windows: portable processing and tests only")
endif()
if(BUILD_TESTING)
  add_executable(patchy_tests tests/core_tests.cpp)
  target_link_libraries(patchy_tests PRIVATE patchy_core)
  add_test(NAME patchy_core COMMAND patchy_tests)
  add_executable(patchy_channel_map_tests tests/channel_map_tests.cpp)
  target_link_libraries(patchy_channel_map_tests PRIVATE patchy_core)
  add_test(NAME patchy_channel_mapping COMMAND patchy_channel_map_tests)
  if(WIN32)
    add_test(NAME patchy_cli_help COMMAND patchy --help)
    add_test(NAME patchy_cli_validation COMMAND ${CMAKE_COMMAND}
      "-DPATCHY_EXE=$<TARGET_FILE:patchy>" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.cmake")
  endif()
endif()

option(PATCHY_BUILD_HARDWARE_TESTS "Build opt-in audible Windows signal smoke test" OFF)
if(WIN32 AND PATCHY_BUILD_HARDWARE_TESTS)
  add_executable(patchy_signal_test tests/windows_signal_test.cpp)
  target_include_directories(patchy_signal_test PRIVATE src/windows)
  target_link_libraries(patchy_signal_test PRIVATE patchy_windows)
  if(MSVC)
    target_compile_options(patchy_signal_test PRIVATE /W4 /permissive- /utf-8)
  endif()
endif()

if(BUILD_TESTING)
  add_executable(patchy_allocation_test tests/allocation_test.cpp)
  target_link_libraries(patchy_allocation_test PRIVATE patchy_core)
  add_test(NAME patchy_prepared_processing_allocations COMMAND patchy_allocation_test)
endif()
